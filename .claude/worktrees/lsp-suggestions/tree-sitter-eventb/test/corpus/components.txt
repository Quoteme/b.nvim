Machine with refined and extended events
================================================================================

machine M1 refines M0 sees C1
variables x y
invariants
  @inv1 x ∈ ℕ
  @inv2 y ⊆ PERSONS
variant limit − x
events
  event INITIALISATION extends INITIALISATION
    then
      @act2 y ≔ ∅
  end

  convergent event step refines inc
    any p q
    where
      @grd1 p ∈ PERSONS ∖ y
      @grd2 x < limit
    with
      @n n = x + 1
    then
      @act1 x, y ≔ x + 1, y ∪ {p}
  end

  event pick
    when
      @grd1 y ≠ ∅
    begin
      @act1 x :∈ 0 ‥ limit
      @act2 y :∣ y' ⊆ y
  end
end
