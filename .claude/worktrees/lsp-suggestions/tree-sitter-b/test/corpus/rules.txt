RULES_MACHINE Checks
REFERENCES Base
OPERATIONS
  COMPUTATION COMP_Numbers
  BODY
    DEFINE Limit TYPE INTEGER DUMMY_VALUE -1 VALUE 100 END
  END;
  RULE RULE_Even
  DEPENDS_ON_COMPUTATION COMP_Numbers
  ERROR_TYPES 2
  ACTIVATION Limit > 0
  BODY
    RULE_FORALL x
    WHERE x : 0..Limit & x mod 2 = 0
    EXPECT x : EvenNumbers
    ERROR_TYPE 1
    COUNTEREXAMPLE STRING_FORMAT("~w is odd", x)
    END;
    RULE_FAIL y
    WHEN y : Odd
    COUNTEREXAMPLE "odd"
    END
  END;
  FUNCTION out <-- FUNC_Primes(n)
  PRECONDITION n : NATURAL
  BODY
    FOR i IN 1..n DO out := out \/ {i} END
  END
END
